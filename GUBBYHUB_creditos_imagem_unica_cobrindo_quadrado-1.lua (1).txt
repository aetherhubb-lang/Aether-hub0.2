    local function resolverImagem(id)
        local candidatos = {
            -- Tenta primeiro carregar o asset diretamente; depois usa thumbnails como fallback.
            "rbxassetid://" .. id,
            "rbxthumb://type=Asset&id=" .. id .. "&w=420&h=420",
            "rbxthumb://type=Asset&id=" .. id .. "&w=150&h=150",
            "https://www.roblox.com/asset-thumbnail/image?assetId=" .. id .. "&width=420&height=420&format=png",
        }
        -- se o ID for de Decal, usa o ID real da imagem como primeira opção
        pcall(function()
            local obj = game:GetObjects("rbxassetid://" .. id)[1]
            if obj then
                local tex = (obj:IsA("Decal") or obj:IsA("Texture")) and obj.Texture
                    or ((obj:IsA("ImageLabel") or obj:IsA("ImageButton")) and obj.Image)
                if tex and tex ~= "" then table.insert(candidatos, 1, tex) end
                obj:Destroy()
            end
        end)
        for _, url in ipairs(candidatos) do
            teste.Image = url
            pcall(function() ContentProvider:PreloadAsync({teste}) end)
            local t0 = tick()
            while not teste.IsLoaded and tick() - t0 < 4 do task.wait(0.1) end
            if teste.IsLoaded then return url end
        end
        warn("⚠️ Não consegui carregar a imagem " .. id .. " (ID inválido, privado ou moderado).")
        return "rbxthumb://type=Asset&id=" .. id .. "&w=420&h=420"
    end

    local ids, vistos = {}, {}
    for _, k in ipairs({"IMG_CREDITOS", "IMG_ABRIR", "IMG_INTERFACE", "IMG_BASE"}) do
        local id = CONFIG[k]
        if not vistos[id] then vistos[id] = true; table.insert(ids, id) end
    end

    local cache = {}
    for i, id in ipairs(ids) do
        status.Text = "Carregando imagens... " .. (i - 1) .. "/" .. #ids
        cache[id] = resolverImagem(id)
        status.Text = "Carregando imagens... " .. i .. "/" .. #ids
        barFill.Size = UDim2.new(i / #ids, 0, 1, 0)
    end

    IMG.Creditos  = cache[CONFIG.IMG_CREDITOS]
    IMG.Abrir     = cache[CONFIG.IMG_ABRIR]
    IMG.Interface = cache[CONFIG.IMG_INTERFACE]
    IMG.Base      = cache[CONFIG.IMG_BASE]

    status.Text = "Pronto!"
    task.wait(0.25)
    gui:Destroy()
end

local function f5()
    local v16 = f1()
    local character3, v17, huge
    if not v16 then return nil end
    character3 = LocalPlayer.Character
    v17, huge = nil, math.huge
    local function f6(p6)
        if not f2(p6, character3) then return end
        local hrp = p6:FindFirstChild("HumanoidRootPart")
        if hrp then
            local mag = (v16.Position - hrp.Position).Magnitude
            if mag <= DistanciaDetectcao and mag < huge then huge = mag; v17 = p6 end
        end
    end
    for _, v in ipairs(Players:GetPlayers()) do if v ~= LocalPlayer and v.Character then f6(v.Character) end end
    for _, v in ipairs(workspace:GetDescendants()) do if v:IsA("Model") then f6(v) end end
    return v17
end

local function ColarNasCostas(p3)
    local char = LocalPlayer.Character
    if not char then return end
    local hrp, hum = char:FindFirstChild("HumanoidRootPart"), char:FindFirstChildOfClass("Humanoid")
    local inimigoHrp = p3:FindFirstChild("HumanoidRootPart")
    if not hrp or not hum or not inimigoHrp then return end
    
    hum.AutoRotate = false
    local tempoInicio = tick()
    local connect = nil
    
    connect = RunService.RenderStepped:Connect(function()
        if not p3 or not p3.Parent or not inimigoHrp or tick() - tempoInicio > 0.35 then 
            hum.AutoRotate = true; connect:Disconnect(); return 
        end
        local targetCFrame = inimigoHrp.CFrame * CFrame.new(0, 0, 4.5)
        hrp.CFrame = targetCFrame
    end)
end

local function f4(p3, p4, p5)
    local character2 = LocalPlayer.Character
    if not character2 then return end
    local humanoidRootPart2 = character2:FindFirstChild("HumanoidRootPart")
    local humanoid2 = character2:FindFirstChildOfClass("Humanoid")
    if not humanoidRootPart2 or not humanoid2 or not p3 then return end
    humanoid2.AutoRotate = false
    local v9 = tick()
    local position = humanoidRootPart2.Position
    local v10 = p5 or KokusenSpeed
    local connect = nil
    connect = RunService.RenderStepped:Connect(function()
        if not p3 or not p3.Parent then
            humanoid2.AutoRotate = true; connect:Disconnect(); return
        else
            local v11 = tick()
            local v12 = math.clamp((v11 - v9) / p4, 0, 1)
            if v12 >= 1 then
                -- chegou: encosta perto do alvo
                local cf = p3.CFrame
                humanoidRootPart2.CFrame = CFrame.lookAt(cf.Position - cf.LookVector * KokusenDistFinal, cf.Position)
                humanoid2.AutoRotate = true; connect:Disconnect(); return
            else
                local v13 = 1 - (1 - v12) ^ 3
                local cframe = p3.CFrame
                local position2 = cframe.Position
                local lerp = position:Lerp(position2 - cframe.LookVector * KokusenDistFinal, v13)
                local v14 = math.sin(v13 * math.pi)
                humanoidRootPart2.CFrame = CFrame.lookAt(lerp, position2) -- reto até o alvo, sem desviar pro lado
                return
            end
        end
    end)
end

-- === SISTEMA DE BYPASS ===
local lp = LocalPlayer
local bypassActive = false
local bypassConn = nil

local function ExecutarBypass()
    local char = lp.Character or lp.CharacterAdded:Wait()
    local root = char:WaitForChild("HumanoidRootPart", 5)
    if not root then return false end

    local sucesso = false
    pcall(function()
        local jogadores = Players:GetPlayers()
        for i = #jogadores, 1, -1 do
            if jogadores[i] == lp then table.remove(jogadores, i) end
        end
        if #jogadores > 0 then
            local alvo = jogadores[math.random(1, #jogadores)]
            if alvo and alvo.Character and alvo.Character:FindFirstChild("HumanoidRootPart") then
                local posAlvo = alvo.Character.HumanoidRootPart.Position + Vector3.new(0, 3, 0)
                root.CFrame = CFrame.new(posAlvo)
                task.wait(0.5)
                if root and root.Parent and (root.Position - posAlvo).Magnitude < 25 then
                    sucesso = true
                end
            end
        end
    end)
    return sucesso
end

local function rotaDoVoid()
    pcall(function()
        local char = lp.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        local hum  = char and char:FindFirstChildOfClass("Humanoid")
        if not root or not hum then return end
        hum.PlatformStand = true
        local bv = Instance.new("BodyVelocity")
        bv.MaxForce = Vector3.new(9e6, 9e6, 9e6)
        bv.Velocity = Vector3.new(0, 150, 0)
        bv.Parent = root
        task.wait(1.3)
        bv.Velocity = Vector3.new(-300, 0, 0)
        task.wait(2.4)
        bv.Velocity = Vector3.zero
        hum.PlatformStand = false
        bv:Destroy()
    end)
end

local flyConnection
local function toggleFly(state)
    FlyAtivo = state
    local char = LocalPlayer.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end

    if state then
        local bv = Instance.new("BodyVelocity", hrp)
        bv.Name = "AetherFly"
        bv.MaxForce = Vector3.new(1e5, 1e5, 1e5)
        local bg = Instance.new("BodyGyro", hrp)
        bg.Name = "AetherGyro"
        bg.P = 1e4; bg.MaxTorque = Vector3.new(1e5, 1e5, 1e5); bg.CFrame = hrp.CFrame

        flyConnection = RunService.RenderStepped:Connect(function()
            if not FlyAtivo then return end
            local cam = workspace.CurrentCamera
            local dir = Vector3.new(0,0,0)
            
            if UserInputService:IsKeyDown(Enum.KeyCode.W) or UserInputService:IsKeyDown(Enum.KeyCode.Up) then dir += cam.CFrame.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) or UserInputService:IsKeyDown(Enum.KeyCode.Down) then dir -= cam.CFrame.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) or UserInputService:IsKeyDown(Enum.KeyCode.Left) or UserInputService:IsKeyDown(Enum.KeyCode.Comma) then dir -= cam.CFrame.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) or UserInputService:IsKeyDown(Enum.KeyCode.Right) or UserInputService:IsKeyDown(Enum.KeyCode.Period) then dir += cam.CFrame.RightVector end
            
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir += Vector3.new(0,1,0) end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then dir -= Vector3.new(0,1,0) end
            
            if dir.Magnitude > 0 then bv.Velocity = dir.Unit * FlySpeed else bv.Velocity = Vector3.new(0,0,0) end
            bg.CFrame = cam.CFrame
        end)
    else
        if flyConnection then flyConnection:Disconnect() end
        if hrp:FindFirstChild("AetherFly") then hrp.AetherFly:Destroy() end
        if hrp:FindFirstChild("AetherGyro") then hrp.AetherGyro:Destroy() end
    end
end

-- ABA ANTI MORTE 💀

local AntiMorteAtivo = false
local AM_VIDA_VERMELHA = 0.50   -- voa quando a vida ficar abaixo de 50% (ajustável no slider)
local AM_VIDA_VERDE    = 0.90   -- só desce quando a vida chegar em 90% ou mais (ajustável no slider)
local AM_DESTINO = CFrame.new(38.5864, 183.0812, 139.7481, 0.999912, -0.000000, -0.013264, 0.000000, 1.000000, -0.000000, 0.013264, 0.000000, 0.999912)
local AM_VELOCIDADE_VOO = 250   -- velocidade do voo (studs por segundo)
local AM_ALTURA_EXTRA   = 150   -- quanto sobe (acima do ponto mais alto) antes de ir pro destino
local AM_TAMANHO_PLAT   = 300   -- tamanho da plataforma (largura x comprimento)
local AM_Forcar = false       -- modo teste: voa agora, sem depender da vida
local AM_NoCeu = false
local AM_Fase = "subindo"       -- "subindo" -> "indo" -> chegou
local AM_Chegou = false
local AM_AlturaAlvo = 0
local AM_PosOriginal = nil
local AM_Plat = nil

local function AM_LimparPlat()
    if AM_Plat then pcall(function() AM_Plat:Destroy() end) end
    AM_Plat = nil
end

-- voo de verdade: BodyVelocity + BodyGyro (sem pular, sem teleportar)
local function AM_PegarVoo(hrp)
    local bv = hrp:FindFirstChild("AetherAMBV")
    if not bv then
        bv = Instance.new("BodyVelocity")
        bv.Name = "AetherAMBV"
        bv.MaxForce = Vector3.new(1e6, 1e6, 1e6)
        bv.Velocity = Vector3.new(0, 0, 0)
        bv.Parent = hrp
    end
    local bg = hrp:FindFirstChild("AetherAMBG")
    if not bg then
        bg = Instance.new("BodyGyro")
        bg.Name = "AetherAMBG"
        bg.P = 1e4
        bg.MaxTorque = Vector3.new(1e6, 1e6, 1e6)
        bg.CFrame = hrp.CFrame
        bg.Parent = hrp
    end
    return bv, bg
end

local function AM_SoltarVoo(hrp)
    if not hrp then return end
    local bv = hrp:FindFirstChild("AetherAMBV"); if bv then bv:Destroy() end
    local bg = hrp:FindFirstChild("AetherAMBG"); if bg then bg:Destroy() end
    local hum = hrp.Parent and hrp.Parent:FindFirstChildOfClass("Humanoid")
    if hum then hum.PlatformStand = false end
end

local function AM_Descer(hrp)
    AM_NoCeu = false
    AM_Chegou = false
    AM_LimparPlat()
    if hrp then
        AM_SoltarVoo(hrp)
        hrp.Anchored = false
        hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
        if AM_PosOriginal then hrp.CFrame = AM_PosOriginal end
    end
    AM_PosOriginal = nil
end

local function AM_Subir(hrp, hum)
    AM_NoCeu = true
    AM_Chegou = false
    AM_Fase = "subindo"
    AM_PosOriginal = hrp.CFrame
    AM_AlturaAlvo = math.max(hrp.Position.Y, AM_DESTINO.Position.Y) + AM_ALTURA_EXTRA
    hum.PlatformStand = true -- sem animação de pulo
    AM_PegarVoo(hrp)
end

local function AM_CriarPlataforma()
    pcall(function()
        local plat = Instance.new("Part")
        plat.Name = "AetherAntiMorte"
        plat.Size = Vector3.new(AM_TAMANHO_PLAT, 4, AM_TAMANHO_PLAT)
        plat.Anchored = true
        plat.CanCollide = true
        plat.Material = Enum.Material.Neon
        plat.Color = Color3.fromRGB(140, 60, 255)
        plat.Transparency = 0.3
        plat.Position = AM_DESTINO.Position - Vector3.new(0, 5, 0)
        plat.Parent = workspace
        AM_Plat = plat
    end)
end

local AM_Erro = nil
local AM_UltPos = nil

-- move o personagem com velocidade; se o jogo ignorar o BodyVelocity, empurra pelo CFrame
local function AM_Mover(hrp, bv, v, dt)
    bv.Velocity = v
    hrp.AssemblyLinearVelocity = v
    if AM_UltPos and v.Magnitude > 0 then
        local andou = (hrp.Position - AM_UltPos).Magnitude
        if andou < v.Magnitude * dt * 0.3 then
            hrp.CFrame = hrp.CFrame + v * dt
        end
    end
    AM_UltPos = hrp.Position
end

local function AM_Passo(dt)
    if not (AntiMorteAtivo or AM_Forcar) then return end
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hum or not hrp or hum.MaxHealth <= 0 then return end

    -- morreu: só limpa o estado (o personagem novo começa do zero)
    if hum.Health <= 0 then
        AM_NoCeu = false; AM_PosOriginal = nil; AM_Chegou = false; AM_UltPos = nil
        AM_LimparPlat()
        return
    end

    local pct = hum.Health / hum.MaxHealth

    if not AM_NoCeu then
        if AM_Forcar or pct <= AM_VIDA_VERMELHA then
            AM_Subir(hrp, hum)
            AM_UltPos = hrp.Position
        end
        return
    end

    if not AM_Chegou then
        local bv, bg = AM_PegarVoo(hrp)
        hum.PlatformStand = true
        hrp.Anchored = false
        if AM_Fase == "subindo" then
            -- 1) voa reto pra cima
            AM_Mover(hrp, bv, Vector3.new(0, AM_VELOCIDADE_VOO, 0), dt)
            if hrp.Position.Y >= AM_AlturaAlvo then AM_Fase = "indo" end
        else
            -- 2) voa até o CFrame do destino
            local falta = AM_DESTINO.Position - hrp.Position
            if falta.Magnitude <= 3 then
                bv.Velocity = Vector3.new(0, 0, 0)
                hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                hrp.CFrame = AM_DESTINO
                bg.CFrame = AM_DESTINO
                AM_Chegou = true
                AM_CriarPlataforma()
            else
                local vel = math.min(AM_VELOCIDADE_VOO, falta.Magnitude * 6 + 10)
                AM_Mover(hrp, bv, falta.Unit * vel, dt)
            end
        end
    elseif AM_Plat and AM_Plat.Parent then
        -- plataforma apareceu: solta o voo e fica em cima dela
        AM_SoltarVoo(hrp)
        if hrp.Position.Y < AM_Plat.Position.Y then
            hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            hrp.CFrame = AM_DESTINO
        end
    else
        -- plataforma não apareceu (ou foi apagada): fica voando parado
        AM_Plat = nil
        local bv = AM_PegarVoo(hrp)
        hum.PlatformStand = true
        bv.Velocity = Vector3.new(0, 0, 0)
        hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
    end

    if not AM_Forcar and pct >= math.max(AM_VIDA_VERDE, AM_VIDA_VERMELHA + 0.1) then AM_Descer(hrp) end
end

RunService.Heartbeat:Connect(function(dt)
    local ok, err = pcall(AM_Passo, dt)
    if not ok then AM_Erro = tostring(err) end
end)

local function registrarESP(model, tipo)
    if espObjs[model] then return end
    local root = model:FindFirstChild("HumanoidRootPart")

    if not root or not hum then return end
    local cor = (tipo == "Dummy") and COR_DUMMY or COR_PLAYER
    local plr = Players:GetPlayerFromCharacter(model)
    local o = { tipo = tipo, model = model, root = root, hum = hum, corBase = cor, player = plr, livre = true, dist = 0, hp = 0 }

    -- amigo? (consulta assíncrona, só uma vez)
    if plr then
        task.spawn(function()
            local ok, r = pcall(function() return LocalPlayer:IsFriendsWith(plr.UserId) end)
            o.amigo = ok and r or false
        end)
    end

    -- Contorno/preenchimento do corpo
    o.hl = Instance.new("Highlight")
    o.hl.Adornee = model; o.hl.FillColor = cor; o.hl.FillTransparency = 0.8

    o.boxFrame.BorderSizePixel = 0; o.boxFrame.Parent = o.box
    o.boxStroke = Instance.new("UIStroke"); o.boxStroke.Color = cor; o.boxStroke.Thickness = 2; o.boxStroke.Parent = o.boxFrame

    o.cantos = {}
    local defs = {
        {0, 0, true}, {0, 0, false}, {1, 0, true}, {1, 0, false},
        {0, 1, true}, {0, 1, false}, {1, 1, true}, {1, 1, false},
    }

        local f = Instance.new("Frame")
        f.AnchorPoint = Vector2.new(d[1], d[2]); f.Position = UDim2.new(d[1], 0, d[2], 0)

    o.barraBg.BackgroundColor3 = Color3.fromRGB(20, 20, 20); o.barraBg.BorderSizePixel = 0; o.barraBg.Parent = o.info

local function removerESP(model)
    local o = espObjs[model]
    if not o then return end
    pcall(function() o.hl:Destroy() end)
    pcall(function() o.box:Destroy() end)
    pcall(function() o.info:Destroy() end)
    apagarDraw(o.line); apagarDraw(o.cabeca); apagarDraw(o.seta)
    if o.ossos then for _, b in ipairs(o.ossos) do apagarDraw(b.line) end end
    espObjs[model] = nil
end

local function escanearPlayers()
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then registrarESP(p.Character, "Player") end
    end
end

local function registrarSeDummy(h)
    local m = h.Parent
    if m and m:IsA("Model") and m ~= LocalPlayer.Character and not Players:GetPlayerFromCharacter(m) then
        registrarESP(m, "Dummy")
    end
end

local function escanearDummies()
    for _, d in ipairs(workspace:GetDescendants()) do
        if d:IsA("Humanoid") then registrarSeDummy(d) end
    end
end

local acumTexto, acumScan = 0, 0
local ultimoErroESP = nil

local function passoESP(dt)
    acumTexto += dt; acumScan += dt
    if acumScan >= 1 then
        acumScan = 0
        if ESPCfg.Player then escanearPlayers() end
    end
    local tick10 = acumTexto >= 0.1
    if tick10 then acumTexto = 0 end

    local cam = workspace.CurrentCamera
    local meuChar = LocalPlayer.Character
    local meuRoot = meuChar and meuChar:FindFirstChild("HumanoidRootPart")
    local agora = tick()
    local centro = cam and Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2) or Vector2.new(0, 0)

    -- FOV e mira
    if fovCirc then
        fovCirc.Visible = ESPCfg.FOV and cam ~= nil
        fovCirc.Position = centro; fovCirc.Radius = ESPCfg.FOVRaio
        fovCirc.Color = Color3.fromRGB(255, 255, 255); fovCirc.Thickness = 1; fovCirc.NumSides = 64; fovCirc.Filled = false
    end
    if miraH and miraV then
        local mostrar = ESPCfg.Mira and cam ~= nil
        miraH.Visible = mostrar; miraV.Visible = mostrar
        miraH.From = centro - Vector2.new(10, 0); miraH.To = centro + Vector2.new(10, 0)
        miraV.From = centro - Vector2.new(0, 10); miraV.To = centro + Vector2.new(0, 10)
        miraH.Color = Color3.fromRGB(255, 255, 255); miraV.Color = Color3.fromRGB(255, 255, 255)
        miraH.Thickness = 1.5; miraV.Thickness = 1.5
    end

    -- 1ª passada: distâncias e alvo mais próximo
    local maisPerto, menor = nil, math.huge
    for model, o in pairs(espObjs) do
        if not model:IsDescendantOf(workspace) or not o.root.Parent or not o.hum.Parent then
            removerESP(model)
        else
            o.dist = meuRoot and (meuRoot.Position - o.root.Position).Magnitude or math.huge
            o.hp = o.hum.Health
            if ESPCfg[o.tipo] and o.hp > 0 and o.dist <= ESPCfg.Max and o.dist < menor then
                menor = o.dist; maisPerto = o
            end
        end
    end

    -- 2ª passada: desenha tudo
    for model, o in pairs(espObjs) do
        local hp = o.hum and o.hum.Parent and o.hum.Health or 0
        local maxhp = o.hum and o.hum.Parent and math.max(o.hum.MaxHealth, 1) or 1
        local dist = o.dist or math.huge
        local ratio = math.clamp(hp / maxhp, 0, 1)

        local visBase = model:IsDescendantOf(workspace) and o.root and o.root.Parent ~= nil
            and ESPCfg[o.tipo] and dist <= ESPCfg.Max and hp > 0
        if visBase and ESPCfg.IgnorarAmigos and o.amigo then visBase = false end

        local v, on = Vector3.new(0, 0, 1), false
        if cam then v, on = cam:WorldToViewportPoint(o.root.Position) end

        local vis = visBase
        if vis and ESPCfg.FOV and cam then
            if not on or (Vector2.new(v.X, v.Y) - centro).Magnitude > ESPCfg.FOVRaio then vis = false end
        end

        -- checagem de parede (a cada 0.1s)
        if vis and ESPCfg.ChecarParede and tick10 and cam then
            local filtro = { model }
            if meuChar then table.insert(filtro, meuChar) end
            rayParams.FilterDescendantsInstances = filtro
            local origem = cam.CFrame.Position
            o.livre = (workspace:Raycast(origem, o.root.Position - origem, rayParams) == nil)
        end

        -- cor final (variante de cor + destaques)
        local cor = o.corBase
        local modo = ESPCfg.ModoCor
        if modo == 2 then cor = Color3.fromHSV((agora * 0.3) % 1, 1, 1)
        elseif modo == 3 then cor = Color3.fromHSV(ratio * 0.33, 1, 1)
        elseif modo == 4 then cor = Color3.fromHSV(math.clamp(dist / math.max(ESPCfg.Max, 1), 0, 1) * 0.6, 1, 1)
        elseif modo == 5 and o.player and o.player.Team then cor = o.player.TeamColor.Color end
        if ESPCfg.ChecarParede and o.livre == false then cor = COR_ATRAS end
        if ESPCfg.DestacarPerto and o == maisPerto then cor = COR_PERTO end
        if ESPCfg.AlertaVidaBaixa and ratio < 0.3 and math.floor(agora * 4) % 2 == 0 then cor = COR_ALERTA end

        -- Contorno / preenchimento
        local hlOn = vis and (ESPCfg.Contorno or ESPCfg.Preench)
        o.hl.Enabled = hlOn
        if hlOn then
            o.hl.OutlineColor = cor; o.hl.FillColor = cor
            o.hl.OutlineTransparency = ESPCfg.Contorno and 0 or 1
            o.hl.FillTransparency = ESPCfg.Preench and ESPCfg.TranspPreench or 1
        end

        -- Caixa + barra lateral
        local lateral = ESPCfg.Vida and ESPCfg.PosBarra == 2
        o.box.Enabled = vis and (ESPCfg.Caixa or lateral)
        if o.box.Enabled then
            o.boxFrame.Visible = ESPCfg.Caixa
            if ESPCfg.Caixa then
                local estilo, esp = ESPCfg.EstiloCaixa, ESPCfg.Espessura
                o.boxFrame.BackgroundColor3 = cor
                o.boxFrame.BackgroundTransparency = (estilo == 3) and math.max(ESPCfg.TranspPreench, 0.5) or 1
                o.boxStroke.Enabled = (estilo ~= 2)
                o.boxStroke.Color = cor; o.boxStroke.Thickness = esp
                for _, c in ipairs(o.cantos) do
                    c.f.Visible = (estilo == 2)
                    if estilo == 2 then
                        c.f.BackgroundColor3 = cor
                        c.f.Size = c.h and UDim2.new(0.25, 0, 0, esp * 1.5) or UDim2.new(0, esp * 1.5, 0.25, 0)
                    end
                end
            end
            o.barraLatBg.Visible = lateral
            if lateral then
                o.barraLat.Size = UDim2.new(1, 0, ratio, 0)
                o.barraLat.BackgroundColor3 = Color3.fromHSV(ratio * 0.33, 1, 1)
            end
        end

        -- Textos (nome, vida, distância, arma, velocidade)
        local barraBaixo = ESPCfg.Vida and ESPCfg.PosBarra == 1
        o.info.Enabled = vis and (ESPCfg.Nome or barraBaixo or ESPCfg.Dist or ESPCfg.Ferramenta or ESPCfg.Velocidade or ESPCfg.Vida)
        if vis and tick10 then
            o.nome.Visible = ESPCfg.Nome
            o.nome.TextColor3 = cor; o.nome.TextSize = ESPCfg.TamTexto
            o.barraBg.Visible = barraBaixo
            o.barra.Size = UDim2.new(ratio, 0, 1, 0)
            o.barra.BackgroundColor3 = Color3.fromHSV(ratio * 0.33, 1, 1)
            local partes = {}
            if ESPCfg.Vida then table.insert(partes, math.floor(hp) .. "/" .. math.floor(maxhp)) end
            if ESPCfg.Dist then table.insert(partes, math.floor(dist) .. "m") end
            o.txt.Visible = #partes > 0
            o.txt.Text = table.concat(partes, "  |  "); o.txt.TextSize = math.max(ESPCfg.TamTexto - 1, 8)
            local extras = {}
            if ESPCfg.Ferramenta then
                local t = model:FindFirstChildOfClass("Tool")
                if t then table.insert(extras, "Arma: " .. t.Name) end
            end
            if ESPCfg.Velocidade then table.insert(extras, "Vel: " .. math.floor(o.root.AssemblyLinearVelocity.Magnitude)) end
            o.txt2.Visible = #extras > 0
            o.txt2.Text = table.concat(extras, "  |  "); o.txt2.TextSize = math.max(ESPCfg.TamTexto - 1, 8)
        end

        -- Tracer
        if o.line then
            local mostrar = false
            if vis and ESPCfg.Tracer and cam and on then
                local vs = cam.ViewportSize
                local origem
                if ESPCfg.OrigemTracer == 1 then origem = Vector2.new(vs.X / 2, vs.Y)
                elseif ESPCfg.OrigemTracer == 2 then origem = centro
                else origem = Vector2.new(vs.X / 2, 0) end
                o.line.From = origem; o.line.To = Vector2.new(v.X, v.Y)
                o.line.Color = cor; o.line.Thickness = math.max(ESPCfg.Espessura * 0.75, 1)
                o.line.Visible = true; mostrar = true
            end
            if not mostrar then o.line.Visible = false end
        end

        -- Esqueleto
        if o.ossos then
            local mostrarEsq = vis and ESPCfg.Esqueleto and cam and on
            for _, bn in ipairs(o.ossos) do
                local ok = false
                if mostrarEsq and bn.a.Parent and bn.b.Parent then
                    local pa, oa = cam:WorldToViewportPoint(bn.a.Position)
                    local pb, ob = cam:WorldToViewportPoint(bn.b.Position)
                    if oa and ob then
                        bn.line.From = Vector2.new(pa.X, pa.Y); bn.line.To = Vector2.new(pb.X, pb.Y)
                        bn.line.Color = cor; bn.line.Thickness = math.max(ESPCfg.Espessura * 0.6, 1)
                        bn.line.Visible = true; ok = true
                    end
                end
                if not ok then bn.line.Visible = false end
            end
        end

        -- Ponto na cabeça
        if o.cabeca then
            local mostrar = false
            local head = model:FindFirstChild("Head")
            if vis and ESPCfg.CabecaPonto and cam and head then
                local pc, oc = cam:WorldToViewportPoint(head.Position)
                if oc then
                    o.cabeca.Position = Vector2.new(pc.X, pc.Y)
                    o.cabeca.Radius = math.clamp(250 / math.max(dist, 5), 3, 14)
                    o.cabeca.Color = cor; o.cabeca.Visible = true; mostrar = true
                end
            end
            if not mostrar then o.cabeca.Visible = false end
        end

        -- Seta apontando pra quem está fora da tela
        if o.seta then
            local mostrar = false
            if visBase and ESPCfg.SetasForaTela and cam and not on then
                local dir = Vector2.new(v.X, v.Y) - centro
                if v.Z < 0 then dir = -dir end
                if dir.Magnitude < 1 then dir = Vector2.new(0, -1) end
                dir = dir.Unit
                local perp = Vector2.new(-dir.Y, dir.X)
                local p = centro + dir * (math.min(centro.X, centro.Y) * 0.8)
                o.seta.PointA = p + dir * 14
                o.seta.PointB = p - dir * 8 + perp * 9
                o.seta.PointC = p - dir * 8 - perp * 9
                o.seta.Color = cor; o.seta.Filled = true
                o.seta.Visible = true; mostrar = true
            end
            if not mostrar then o.seta.Visible = false end
        end
    end
end

table.insert(espConns, RunService.Heartbeat:Connect(function(dt)
    local ok, err = pcall(passoESP, dt)
    if not ok and err ~= ultimoErroESP then
        ultimoErroESP = err
        warn("[Aether ESP] " .. tostring(err))
    end
end))

end)

local function limparExt(e)
    for _, k in ipairs({"info2", "hl2", "trail", "att0", "att1", "anel", "aura", "cabeca"}) do
        if e[k] then pcall(function() e[k]:Destroy() end) end
    end
    if e.cubo then for _, l in pairs(e.cubo) do apagarDraw(l) end end
    apagarDraw(e.visao)
end

local function esconderExt(e)
    e.info2.Enabled = false; e.hl2.Enabled = false; e.trail.Enabled = false
    e.anel.Visible = false; e.aura.Enabled = false
    if e.cabeca then e.cabeca.Visible = false end
    if e.cubo then for _, l in pairs(e.cubo) do l.Visible = false end end
    if e.visao then e.visao.Visible = false end
end

local acum2 = 0
local ultimoErroESP2 = nil

local function passoESP2(dt)
    acum2 += dt
    local tick10 = acum2 >= 0.1
    if tick10 then acum2 = 0 end

    local cam = workspace.CurrentCamera
    local meuChar = LocalPlayer.Character
    local meuRoot = meuChar and meuChar:FindFirstChild("HumanoidRootPart")

    for model, e in pairs(ext) do
        if not espObjs[model] then limparExt(e); ext[model] = nil end
    end
    for m, d in pairs(radarDots) do
        if not espObjs[m] then d:Destroy(); radarDots[m] = nil else d.Visible = false end
    end

    local algumPorModelo = ESP2.Username or ESP2.UserId or ESP2.Idade or ESP2.Time or ESP2.Estado
        or ESP2.WalkSpeed or ESP2.JumpPower or ESP2.VidaPct or ESP2.Altura or ESP2.TagAmigo
        or ESP2.Caixa3D or ESP2.LinhaVisao or ESP2.ChamsDuplo or ESP2.Rastro or ESP2.Anel
        or ESP2.Aura or ESP2.CaixaCabeca

    Radar.Visible = ESP2.Radar
    local total, perto, lista = 0, 0, {}

    for model, o in pairs(espObjs) do
        local hum, root = o.hum, o.root
        if root and root.Parent and hum and hum.Parent and model:IsDescendantOf(workspace) then
            local dist = meuRoot and (root.Position - meuRoot.Position).Magnitude or math.huge
            local vis = ESPCfg[o.tipo] and dist <= ESPCfg.Max and hum.Health > 0
            if vis and ESPCfg.IgnorarAmigos and o.amigo then vis = false end

            if vis then
                total += 1
                if dist <= 100 then perto += 1 end
                lista[#lista + 1] = { n = o.nome and o.nome.Text or model.Name, d = dist }

                -- radar
                if ESP2.Radar and cam then
                    local rel = cam.CFrame:PointToObjectSpace(root.Position)
                    local escala = RAIO_RADAR_PX / math.max(ESP2.RaioRadar, 1)
                    local px, py = rel.X * escala, rel.Z * escala
                    local m = math.sqrt(px * px + py * py)
                    if m > RAIO_RADAR_PX - 4 then
                        local f = (RAIO_RADAR_PX - 4) / m
                        px, py = px * f, py * f
                    end
                    local d = radarDots[model] or pontoRadar(model)
                    d.Position = UDim2.new(0.5, px - 3, 0.5, py - 3)
                    d.BackgroundColor3 = o.corBase; d.Visible = true
                end
            end

            local e = ext[model]
            if not e and vis and algumPorModelo then e = criarExt(o) end

            if e then
                if not (vis and algumPorModelo) then
                    esconderExt(e)
                else
                    local cor = o.corBase
                    e.hl2.Enabled = ESP2.ChamsDuplo
                    e.trail.Enabled = ESP2.Rastro
                    e.anel.Visible = ESP2.Anel
                    e.aura.Enabled = ESP2.Aura
                    if e.cabeca then e.cabeca.Visible = ESP2.CaixaCabeca end

                    if tick10 then
                        e.trail.Color = ColorSequence.new(cor)
                        e.anel.Color3 = cor; e.aura.Color = cor
                        if e.cabeca then e.cabeca.Color3 = cor end
                        local plr = o.player
                        local L = {}
                        if ESP2.TagAmigo and o.amigo then table.insert(L, "[AMIGO]") end
                        if ESP2.Username and plr then table.insert(L, "@" .. plr.Name) end
                        if ESP2.UserId and plr then table.insert(L, "ID: " .. plr.UserId) end
                        if ESP2.Idade and plr then table.insert(L, "Conta: " .. plr.AccountAge .. " dias") end
                        if ESP2.Time then table.insert(L, "Time: " .. (plr and plr.Team and plr.Team.Name or "Sem time")) end
                        if ESP2.Estado then
                            local nomeEst = tostring(hum:GetState()):match("%.(%w+)$") or "?"
                            table.insert(L, "Estado: " .. (ESTADOS_PT[nomeEst] or nomeEst))
                        end
                        if ESP2.VidaPct then
                            table.insert(L, "Vida: " .. math.floor(hum.Health / math.max(hum.MaxHealth, 1) * 100) .. "%")
                        end
                        if ESP2.WalkSpeed then table.insert(L, "Speed: " .. math.floor(hum.WalkSpeed)) end
                        if ESP2.JumpPower then
                            local pulo = hum.UseJumpPower and hum.JumpPower or hum.JumpHeight
                            table.insert(L, "Pulo: " .. math.floor(pulo))
                        end
                        if ESP2.Altura then table.insert(L, "Altura: " .. math.floor(root.Position.Y) .. " Y") end
                        e.txt.TextColor3 = cor
                        e.txt.Text = table.concat(L, "\n")
                        e.info2.Enabled = #L > 0
                    end

                    -- caixa 3D
                    if e.cubo then
                        local ok3d = ESP2.Caixa3D and cam ~= nil
                        local pts = {}
                        if ok3d then
                            for i, off in ipairs(CANTOS3D) do
                                local v = cam:WorldToViewportPoint(root.CFrame:PointToWorldSpace(off))
                                if v.Z <= 0 then ok3d = false; break end
                                pts[i] = Vector2.new(v.X, v.Y)
                            end
                        end
                        for i, par in ipairs(ARESTAS3D) do
                            local l = e.cubo[i]
                            if l then
                                if ok3d then
                                    l.From = pts[par[1]]; l.To = pts[par[2]]
                                    l.Color = cor; l.Thickness = ESPCfg.Espessura; l.Visible = true
                                else
                                    l.Visible = false
                                end
                            end
                        end
                    end

                    -- linha de visão
                    if e.visao then
                        local okv = false
                        if ESP2.LinhaVisao and cam then
                            local base = e.head or root
                            local p1 = base.Position
                            local p2 = p1 + base.CFrame.LookVector * 10
                            local a, b = cam:WorldToViewportPoint(p1), cam:WorldToViewportPoint(p2)
                            if a.Z > 0 and b.Z > 0 then
                                e.visao.From = Vector2.new(a.X, a.Y); e.visao.To = Vector2.new(b.X, b.Y)
                                e.visao.Color = Color3.fromRGB(255, 255, 0)
                                e.visao.Thickness = math.max(ESPCfg.Espessura * 0.75, 1)
                                e.visao.Visible = true; okv = true
                            end
                        end
                        if not okv then e.visao.Visible = false end
                    end
                end
            end
        end
    end

    -- contador e lista na tela
    LabelContador.Visible = ESP2.Contador
    if ESP2.Contador and tick10 then
        LabelContador.Text = "👥 Alvos: " .. total .. "  |  ≤100m: " .. perto
    end
    LabelLista.Visible = ESP2.ListaPerto
    if ESP2.ListaPerto and tick10 then
        table.sort(lista, function(a, b) return a.d < b.d end)
        local linhas = { "  Mais perto:" }
        for i = 1, math.min(5, #lista) do
            table.insert(linhas, string.format("  %d. %s — %dm", i, lista[i].n, math.floor(lista[i].d)))
        end
        LabelLista.Text = table.concat(linhas, "\n")
    end
end

    end
end))

local dragging, dragInput, dragStart, startPos
local function makeDraggable(dragArea)
    dragArea.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true; dragStart = input.Position; startPos = MainFrame.Position
            input.Changed:Connect(function() if input.UserInputState == Enum.UserInputState.End then dragging = false end end)
        end
    end)
    dragArea.InputChanged:Connect(function(input) if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then dragInput = input end end)
end

makeDraggable(TopBar); makeDraggable(MiniPanel)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        MainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

local isMinimized = false
local function Minimizar()
    isMinimized = true
    TopBar.Visible = false; TabBar.Visible = false; ContentArea.Visible = false; MiniPanel.Visible = true
    TweenService:Create(MainFrame, TweenInfo.new(0.3, Enum.EasingStyle.Back), {Size = TAMANHO_MINI}):Play()
end

local function Restaurar()
    isMinimized = false; MiniPanel.Visible = false
    TweenService:Create(MainFrame, TweenInfo.new(0.3, Enum.EasingStyle.Back), {Size = TAMANHO_NORMAL}):Play()
    task.wait(0.1)
    TopBar.Visible = true; TabBar.Visible = true; ContentArea.Visible = true
end

MinBtn.MouseButton1Click:Connect(function() if not isMinimized then Minimizar() end end)
MiniPanel.MouseButton1Click:Connect(function() if isMinimized then Restaurar() end end)
